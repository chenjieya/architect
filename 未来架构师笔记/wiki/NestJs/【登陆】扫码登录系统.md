扫码登录已经成为现代 Web 应用的标准功能之一，它提供了比传统账号密码登录更便捷、更安全的用户体验。今天，我将分享如何使用 NestJS 框架构建一个完整的企业级扫码登录系统。

## 1. 🎯 扫码登录的核心原理

扫码登录的本质是一种**设备授权**机制，其核心流程包括：

1. **生成二维码**：服务端生成唯一标识（UUID）并嵌入二维码
2. **等待扫描**：用户使用已登录设备扫描二维码
3. **确认登录**：用户在已登录设备上确认授权
4. **完成登录**：新设备获取登录凭证，完成登录

### 4.2 Redis 服务封装

```typescript
// src/redis/redis.service.ts
import { Injectable } from "@nestjs/common";
import Redis from "ioredis";

@Injectable()
export class RedisService {
  private readonly redis: Redis;

  constructor() {
    this.redis = new Redis({
      host: process.env.REDIS_HOST || "localhost",
      port: parseInt(process.env.REDIS_PORT || "6379"),
      db: parseInt(process.env.REDIS_DB || "0"),
    });
  }

  // 存储二维码数据
  async setQRCode(id: string, data: any, ttl: number = 300): Promise<void> {
    await this.redis.setex(`qrcode:${id}`, ttl, JSON.stringify(data));
  }

  // 获取二维码数据
  async getQRCode(id: string): Promise<any> {
    const data = await this.redis.get(`qrcode:${id}`);
    return data ? JSON.parse(data) : null;
  }

  // 更新二维码状态
  async updateQRCodeStatus(id: string, status: string): Promise<void> {
    const key = `qrcode:${id}`;
    const data = await this.getQRCode(id);
    if (data) {
      data.status = status;
      data.updatedAt = new Date().toISOString();
      const ttl = await this.redis.ttl(key);
      await this.redis.setex(key, ttl > 0 ? ttl : 300, JSON.stringify(data));
    }
  }
}
```

### 4.3 二维码服务实现

```typescript
// src/qrcode/qrcode.service.ts
import { Injectable, NotFoundException } from "@nestjs/common";
import { v4 as uuidv4 } from "uuid";
import * as QRCode from "qrcode";
import { RedisService } from "../redis/redis.service";
import { QRCodeStatus } from "./entities/qrcode.entity";
import { EventEmitter2 } from "@nestjs/event-emitter";

@Injectable()
export class QrCodeService {
  private readonly QR_CODE_EXPIRE_TIME = 300; // 5分钟过期

  constructor(
    private readonly redisService: RedisService,
    private readonly eventEmitter: EventEmitter2
  ) {}

  // 生成二维码
  async generateQRCode(deviceInfo?: any): Promise<{
    qrcodeId: string;
    qrcodeUrl: string;
    expiresAt: Date;
  }> {
    const qrcodeId = uuidv4();
    const token = uuidv4();
    const createdAt = new Date();
    const expiresAt = new Date(Date.now() + this.QR_CODE_EXPIRE_TIME * 1000);

    const qrCodeData = {
      id: qrcodeId,
      token,
      status: QRCodeStatus.PENDING,
      createdAt,
      expiresAt,
      deviceInfo,
    };

    // 存储到Redis
    await this.redisService.setQRCode(
      qrcodeId,
      qrCodeData,
      this.QR_CODE_EXPIRE_TIME
    );

    // 生成二维码URL
    const qrcodeUrl = await QRCode.toDataURL(
      JSON.stringify({
        id: qrcodeId,
        token,
        type: "login",
        timestamp: createdAt.getTime(),
      })
    );

    return {
      qrcodeId,
      qrcodeUrl,
      expiresAt,
    };
  }

  // 扫描二维码
  async scanQRCode(id: string, user: any): Promise<void> {
    const qrCodeData = await this.redisService.getQRCode(id);

    if (!qrCodeData) {
      throw new NotFoundException("二维码不存在或已过期");
    }

    if (qrCodeData.status !== QRCodeStatus.PENDING) {
      throw new Error("二维码状态异常");
    }

    // 更新状态为已扫描
    qrCodeData.status = QRCodeStatus.SCANNED;
    qrCodeData.userId = user.id;
    qrCodeData.scannedAt = new Date();

    await this.redisService.setQRCode(id, qrCodeData, this.QR_CODE_EXPIRE_TIME);

    // 发送扫描事件
    this.eventEmitter.emit("qrcode.scanned", {
      qrcodeId: id,
      userId: user.id,
      userInfo: {
        username: user.username,
        avatar: user.avatar,
      },
    });
  }

  // 确认登录
  async confirmQRCode(id: string, user: any): Promise<{ accessToken: string }> {
    const qrCodeData = await this.redisService.getQRCode(id);

    if (!qrCodeData) {
      throw new NotFoundException("二维码不存在或已过期");
    }

    if (qrCodeData.status !== QRCodeStatus.SCANNED) {
      throw new Error("二维码状态异常");
    }

    if (qrCodeData.userId !== user.id) {
      throw new Error("用户不匹配");
    }

    // 更新状态为已确认
    qrCodeData.status = QRCodeStatus.CONFIRMED;
    qrCodeData.confirmedAt = new Date();

    await this.redisService.setQRCode(
      id,
      qrCodeData,
      60 // 确认后保留1分钟
    );

    // 生成JWT token
    const accessToken = this.generateJWT(user);

    // 发送确认事件
    this.eventEmitter.emit("qrcode.confirmed", {
      qrcodeId: id,
      userId: user.id,
      accessToken,
    });

    return { accessToken };
  }

  // 检查二维码状态
  async checkQRCodeStatus(id: string): Promise<{
    status: QRCodeStatus;
    userInfo?: any;
    accessToken?: string;
  }> {
    const qrCodeData = await this.redisService.getQRCode(id);

    if (!qrCodeData) {
      return { status: QRCodeStatus.EXPIRED };
    }

    return {
      status: qrCodeData.status,
      userInfo: qrCodeData.userId
        ? {
            id: qrCodeData.userId,
            // 这里可以从数据库获取完整的用户信息
          }
        : undefined,
      accessToken: qrCodeData.accessToken,
    };
  }

  private generateJWT(user: any): string {
    // JWT生成逻辑
    // 实际实现中会使用jsonwebtoken库
    return `jwt-token-for-${user.id}`;
  }
}
```

### 4.4 WebSocket 网关实现

```typescript
// src/websocket/websocket.gateway.ts
import {
  WebSocketGateway,
  WebSocketServer,
  SubscribeMessage,
  OnGatewayConnection,
  OnGatewayDisconnect,
} from "@nestjs/websockets";
import { Server, Socket } from "socket.io";
import { EventEmitter2 } from "@nestjs/event-emitter";

@WebSocketGateway({
  cors: {
    origin: "*",
  },
  namespace: "/qrcode",
})
export class QrCodeGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer()
  server: Server;

  // 存储连接映射：qrcodeId -> socketId
  private qrCodeConnections = new Map<string, string>();

  constructor(private readonly eventEmitter: EventEmitter2) {
    // 监听二维码扫描事件
    this.eventEmitter.on("qrcode.scanned", (data) => {
      this.handleQrCodeScanned(data);
    });

    // 监听二维码确认事件
    this.eventEmitter.on("qrcode.confirmed", (data) => {
      this.handleQrCodeConfirmed(data);
    });
  }

  handleConnection(client: Socket) {
    console.log(`Client connected: ${client.id}`);
  }

  handleDisconnect(client: Socket) {
    // 清理连接映射
    for (const [qrcodeId, socketId] of this.qrCodeConnections) {
      if (socketId === client.id) {
        this.qrCodeConnections.delete(qrcodeId);
        break;
      }
    }
  }

  @SubscribeMessage("subscribe_qrcode")
  handleSubscribe(client: Socket, payload: { qrcodeId: string }) {
    const { qrcodeId } = payload;

    // 保存映射关系
    this.qrCodeConnections.set(qrcodeId, client.id);

    // 发送订阅成功消息
    client.emit("subscribe_success", {
      qrcodeId,
      message: "成功订阅二维码状态",
    });
  }

  @SubscribeMessage("check_status")
  async handleCheckStatus(client: Socket, payload: { qrcodeId: string }) {
    // 检查二维码状态并返回
    // ...
  }

  private handleQrCodeScanned(data: any) {
    const socketId = this.qrCodeConnections.get(data.qrcodeId);
    if (socketId) {
      this.server.to(socketId).emit("qrcode_scanned", {
        qrcodeId: data.qrcodeId,
        userInfo: data.userInfo,
        timestamp: new Date().toISOString(),
      });
    }
  }

  private handleQrCodeConfirmed(data: any) {
    const socketId = this.qrCodeConnections.get(data.qrcodeId);
    if (socketId) {
      this.server.to(socketId).emit("qrcode_confirmed", {
        qrcodeId: data.qrcodeId,
        accessToken: data.accessToken,
        timestamp: new Date().toISOString(),
      });

      // 发送成功后移除映射
      this.qrCodeConnections.delete(data.qrcodeId);
    }
  }
}
```

```typescript
// src/qrcode/qrcode.controller.ts
import {
  Controller,
  Get,
  Post,
  Query,
  Body,
  Headers,
  UseGuards,
} from "@nestjs/common";
import { QrCodeService } from "./qrcode.service";
import { AuthGuard } from "../auth/guards/auth.guard";
import { CurrentUser } from "../auth/decorators/current-user.decorator";

@Controller("qrcode")
export class QrCodeController {
  constructor(private readonly qrCodeService: QrCodeService) {}

  @Get("generate")
  async generate(
    @Headers("user-agent") userAgent: string,
    @Headers("x-forwarded-for") ip: string
  ) {
    const deviceInfo = {
      userAgent,
      ip: ip || "unknown",
    };

    return await this.qrCodeService.generateQRCode(deviceInfo);
  }

  @Post("scan")
  @UseGuards(AuthGuard)
  async scan(@Query("id") id: string, @CurrentUser() user: any) {
    return await this.qrCodeService.scanQRCode(id, user);
```

## 6. 🛡️ 安全考虑与最佳实践

### 6.1 **防止重放攻击**

- 二维码 token 一次性有效
- 设置合理的过期时间（建议 5 分钟）
- 使用非对称加密签名

### 6.2 **防止跨站请求伪造**

- 验证请求来源
- 使用 CORS 策略限制域名
- 添加 CSRF token 验证

### 6.3 **保护用户隐私**

- 不传输敏感信息到二维码
- 使用临时 token 而非永久凭证
- 记录操作日志用于审计

### 6.4 **性能优化**

- Redis 缓存二维码数据
- WebSocket 替代轮询
- 连接池管理和资源清理
