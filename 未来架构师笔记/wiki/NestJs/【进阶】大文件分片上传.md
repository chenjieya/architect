在现代 Web 应用中，文件上传是一个非常常见的功能。然而，当涉及到处理大文件（如视频、高清图片、大型数据集等）时，传统的文件上传方式会面临诸多挑战。本文将深入探讨大文件分片上传的技术原理、实现方案以及实际应用中的最佳实践。

## 1. 传统文件上传的局限性

在标准的 HTTP 文件上传中，当用户选择一个大文件进行上传时：

1. **长时间占用连接**：上传过程需要持续的网络连接
2. **网络不稳定风险**：如果网络中断，整个上传需要重新开始
3. **服务器压力大**：一次性处理大文件会占用大量内存
4. **用户体验差**：用户需要长时间等待，无法看到上传进度

以一个 100MB 的文件为例，假设上传速度为 1MB/s，则需要大约 100 秒。如果是 1GB 的文件，则需要约 17 分钟。这在实际应用中是不可接受的。

## 2. 分片上传的技术原理

分片上传（Chunked Upload）通过将大文件分割成多个小块（分片），然后并行或顺序上传这些分片，最后在服务器端重新组合，从而解决了大文件上传的问题。

### 2.1 核心概念

1. **文件分片**：将大文件按固定大小分割成多个小文件
2. **并行上传**：多个分片可以同时上传，提高传输效率
3. **断点续传**：某个分片上传失败只需重传该分片，无需重新开始
4. **分片合并**：所有分片上传完成后在服务器端合并成完整文件

### 3.2 服务端实现

#### 3.2.1 基础文件上传接口

首先，我们创建一个支持多文件上传的基础接口：

```typescript
// src/app.controller.ts
import {
  Controller,
  Post,
  Get,
  UseInterceptors,
  UploadedFiles,
  Body,
  Query,
} from "@nestjs/common";
import { FilesInterceptor } from "@nestjs/platform-express";
import * as fs from "fs";
import * as path from "path";

@Controller()
export class AppController {
  // 文件上传目录
  private readonly uploadDir = "uploads";

  constructor() {
    // 确保上传目录存在
    if (!fs.existsSync(this.uploadDir)) {
      fs.mkdirSync(this.uploadDir, { recursive: true });
    }
  }

  /**
   * 分片上传接口
   */
  @Post("upload")
  @UseInterceptors(
    FilesInterceptor("files", 20, {
      dest: this.uploadDir,
    })
  )
  async uploadFiles(
    @UploadedFiles() files: Array<Express.Multer.File>,
    @Body() body: { name: string }
  ) {
    console.log("收到分片:", body.name);
    console.log("文件信息:", files);

    // 解析文件名和分片索引
    // 格式：randomStr_fileName-chunkIndex
    const matchResult = body.name.match(/^([^_]+)_([^-]+)-(\d+)$/);

    if (!matchResult) {
      throw new Error("文件名格式不正确");
    }

    const [_, randomStr, fileName, chunkIndex] = matchResult;

    // 创建分片存储目录
    const chunkDir = path.join(
      this.uploadDir,
      `chunks_${randomStr}_${fileName}`
    );

    if (!fs.existsSync(chunkDir)) {
      fs.mkdirSync(chunkDir, { recursive: true });
    }

    // 移动分片文件到指定目录
    const chunkPath = path.join(chunkDir, `chunk_${chunkIndex}`);
    fs.copyFileSync(files[0].path, chunkPath);

    // 删除临时文件
    fs.unlinkSync(files[0].path);

    // 返回分片信息
    return {
      success: true,
      message: "分片上传成功",
      data: {
        fileName,
        chunkIndex: parseInt(chunkIndex),
        chunkDir,
        chunkPath,
      },
    };
  }

  /**
   * 分片合并接口
   */
  @Get("merge")
  async merge(@Query("name") name: string) {
    // 解析随机字符串和文件名
    const [randomStr, fileName] = name.split("_", 2);
    const chunkDir = path.join(
      this.uploadDir,
      `chunks_${randomStr}_${fileName}`
    );

    if (!fs.existsSync(chunkDir)) {
      throw new Error("分片目录不存在");
    }

    // 读取所有分片文件
    const chunkFiles = fs
      .readdirSync(chunkDir)
      .filter((file) => file.startsWith("chunk_"))
      .sort((a, b) => {
        const indexA = parseInt(a.split("_")[1]);
        const indexB = parseInt(b.split("_")[1]);
        return indexA - indexB;
      });

    if (chunkFiles.length === 0) {
      throw new Error("没有找到分片文件");
    }

    // 目标文件路径
    const targetFilePath = path.join(this.uploadDir, fileName);
    const writeStream = fs.createWriteStream(targetFilePath);

    // 合并分片的Promise数组
    const mergePromises = chunkFiles.map((chunkFile, index) => {
      return new Promise((resolve, reject) => {
        const chunkPath = path.join(chunkDir, chunkFile);
        const readStream = fs.createReadStream(chunkPath);

        readStream.pipe(writeStream, { end: false });

        readStream.on("end", () => {
          console.log(`分片 ${index} 合并完成`);
          // 删除已合并的分片文件
          fs.unlinkSync(chunkPath);
          resolve(true);
        });

        readStream.on("error", (error) => {
          reject(error);
        });
      });
    });

    // 等待所有分片合并完成
    await Promise.all(mergePromises);

    // 结束写入流
    writeStream.end();

    return new Promise((resolve, reject) => {
      writeStream.on("finish", () => {
        // 删除分片目录
        fs.rmdirSync(chunkDir);

        resolve({
          success: true,
          message: "文件合并成功",
          data: {
            fileName,
            filePath: targetFilePath,
            fileSize: fs.statSync(targetFilePath).size,
          },
        });
      });

      writeStream.on("error", reject);
    });
  }
}
```

#### 3.2.2 添加分片状态检查接口

在实际应用中，我们还需要检查分片的上传状态：

```typescript
@Get('check-chunks')
checkChunks(@Query('name') name: string) {
  const [randomStr, fileName] = name.split('_', 2);
  const chunkDir = path.join(this.uploadDir, `chunks_${randomStr}_${fileName}`);

  if (!fs.existsSync(chunkDir)) {
    return {
      exists: false,
      uploadedChunks: [],
      totalChunks: 0
    };
  }

  const chunkFiles = fs.readdirSync(chunkDir)
    .filter(file => file.startsWith('chunk_'))
    .map(file => parseInt(file.split('_')[1]))
    .sort((a, b) => a - b);

  return {
    exists: true,
    uploadedChunks: chunkFiles,
    totalChunks: chunkFiles.length > 0 ? Math.max(...chunkFiles) + 1 : 0
  };
}
```

```html
    <script>
      // 全局配置
      const CONFIG = {
        CHUNK_SIZE: 2 * 1024 * 1024, // 2MB 分片大小
        MAX_CONCURRENT_UPLOADS: 4, // 最大并发上传数
        SERVER_URL: "http://localhost:3000",
      };

      class FileUploader {
        constructor() {
          this.file = null;
          this.chunks = [];
          this.uploading = false;
          this.cancelled = false;
          this.progress = {
            uploaded: 0,
            total: 0,
          };
          this.chunkStatus = [];
          this.startTime = null;
          this.randomStr = this.generateRandomString(10);

          this.initElements();
          this.bindEvents();
        }

        initElements() {
          this.elements = {
            uploadArea: document.getElementById("uploadArea"),
            fileInput: document.getElementById("fileInput"),
            fileInfo: document.getElementById("fileInfo"),
            fileName: document.getElementById("fileName"),
            fileSize: document.getElementById("fileSize"),
            chunkCount: document.getElementById("chunkCount"),
            chunkSize: document.getElementById("chunkSize"),
            progressContainer: document.getElementById("progressContainer"),
            progressFill: document.getElementById("progressFill"),
            progressText: document.getElementById("progressText"),
            speedText: document.getElementById("speedText"),
            uploadBtn: document.getElementById("uploadBtn"),
            cancelBtn: document.getElementById("cancelBtn"),
            chunksGrid: document.getElementById("chunksGrid"),
            statusMessage: document.getElementById("statusMessage"),
          };
        }

        bindEvents() {
          // 点击上传区域选择文件
          this.elements.uploadArea.addEventListener("click", () => {
            this.elements.fileInput.click();
          });

          // 文件选择事件
          this.elements.fileInput.addEventListener("change", (e) => {
            this.handleFileSelect(e.target.files[0]);
          });

          // 拖拽事件
          this.elements.uploadArea.addEventListener("dragover", (e) => {
            e.preventDefault();
            this.elements.uploadArea.classList.add("dragover");
          });

          this.elements.uploadArea.addEventListener("dragleave", () => {
            this.elements.uploadArea.classList.remove("dragover");
          });

          this.elements.uploadArea.addEventListener("drop", (e) => {
            e.preventDefault();
            this.elements.uploadArea.classList.remove("dragover");
            this.handleFileSelect(e.dataTransfer.files[0]);
          });

          // 按钮事件
          this.elements.uploadBtn.addEventListener("click", () => {
            this.startUpload();
          });

          this.elements.cancelBtn.addEventListener("click", () => {
            this.cancelUpload();
          });
        }

        generateRandomString(length) {
          const chars =
            "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
          let result = "";
          for (let i = 0; i < length; i++) {
            result += chars.charAt(Math.floor(Math.random() * chars.length));
          }
          return result;
        }

        formatFileSize(bytes) {
          if (bytes === 0) return "0 Bytes";
          const k = 1024;
          const sizes = ["Bytes", "KB", "MB", "GB", "TB"];
          const i = Math.floor(Math.log(bytes) / Math.log(k));
          return (
            parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + " " + sizes[i]
          );
        }

        handleFileSelect(file) {
          if (!file) return;

          this.file = file;
          this.cancelled = false;

          // 显示文件信息
          this.elements.fileName.textContent = file.name;
          this.elements.fileSize.textContent = this.formatFileSize(file.size);

          // 计算分片
          this.chunks = this.splitFile(file);
          this.elements.chunkCount.textContent = this.chunks.length;
          this.elements.chunkSize.textContent = this.formatFileSize(
            CONFIG.CHUNK_SIZE
          );

          // 初始化分片状态
          this.chunkStatus = new Array(this.chunks.length).fill("pending");

          // 更新UI
          this.elements.fileInfo.classList.remove("hidden");
          this.elements.progressContainer.classList.remove("hidden");
          this.elements.uploadBtn.disabled = false;
          this.elements.cancelBtn.disabled = false;
          this.elements.statusMessage.classList.add("hidden");

          // 渲染分片状态网格
          this.renderChunksGrid();

          // 重置进度
          this.updateProgress(0);
        }

        splitFile(file) {
          const chunks = [];
          let startPos = 0;

          while (startPos < file.size) {
            const endPos = Math.min(startPos + CONFIG.CUNK_SIZE, file.size);
            const chunk = file.slice(startPos, endPos);
            chunks.push({
              blob: chunk,
              index: chunks.length,
              start: startPos,
              end: endPos,
            });
            startPos = endPos;
          }

          return chunks;
        }

        renderChunksGrid() {
          this.elements.chunksGrid.innerHTML = "";

          this.chunkStatus.forEach((status, index) => {
            const chunkItem = document.createElement("div");
            chunkItem.className = `chunk-item chunk-${status}`;
            chunkItem.textContent = index + 1;
            chunkItem.title = `分片 ${index + 1}: ${status}`;
            this.elements.chunksGrid.appendChild(chunkItem);
          });
        }

        updateChunkStatus(index, status) {
          this.chunkStatus[index] = status;
          this.renderChunksGrid();
        }

        updateProgress(uploadedBytes) {
          this.progress.uploaded = uploadedBytes;
          this.progress.total = this.file.size;

          const percentage = (uploadedBytes / this.file.size) * 100;
          this.elements.progressFill.style.width = `${percentage}%`;
          this.elements.progressText.textContent = `${percentage.toFixed(1)}%`;

          // 计算上传速度
          if (this.startTime) {
            const elapsedSeconds = (Date.now() - this.startTime) / 1000;
            const speed = uploadedBytes / elapsedSeconds;
            this.elements.speedText.textContent = `速度: ${this.formatFileSize(
              speed
            )}/s`;
          }
        }

        async startUpload() {
          if (!this.file || this.uploading) return;

          this.uploading = true;
          this.cancelled = false;
          this.startTime = Date.now();
          this.elements.uploadBtn.disabled = true;
          this.elements.uploadBtn.textContent = "上传中...";

          this.showStatus("开始上传文件...", "info");

          try {
            // 并发上传分片
            const uploadPromises = [];
            const activeUploads = new Set();

            for (let i = 0; i < this.chunks.length; i++) {
              if (this.cancelled) break;

              // 控制并发数
              if (activeUploads.size >= CONFIG.MAX_CONCURRENT_UPLOADS) {
                await Promise.race(activeUploads);
              }

              const uploadPromise = this.uploadChunk(i);
              activeUploads.add(uploadPromise);

              uploadPromise.then(() => {
                activeUploads.delete(uploadPromise);
              });

              uploadPromises.push(uploadPromise);
            }

            await Promise.all(uploadPromises);

            if (this.cancelled) {
              this.showStatus("上传已取消", "error");
              return;
            }

            // 合并文件
            this.showStatus("正在合并文件...", "info");
            await this.mergeFile();

            this.showStatus("文件上传成功！", "success");
          } catch (error) {
            if (!this.cancelled) {
              this.showStatus(`上传失败: ${error.message}`, "error");
            }
          } finally {
            this.uploading = false;
            this.elements.uploadBtn.disabled = false;
            this.elements.uploadBtn.textContent = "开始上传";
          }
        }

        async uploadChunk(index) {
          if (this.cancelled) return;

          this.updateChunkStatus(index, "uploading");

          const chunk = this.chunks[index];
          const formData = new FormData();

          // 构建分片文件名：randomStr_originalFileName-chunkIndex
          const chunkName = `${this.randomStr}_${this.file.name}-${index}`;
          formData.set("name", chunkName);
          formData.append(
            "files",
            chunk.blob,
            `${this.file.name}.part${index}`
          );

          try {
            const response = await axios.post(
              `${CONFIG.SERVER_URL}/upload`,
              formData,
              {
                headers: { "Content-Type": "multipart/form-data" },
                onUploadProgress: (progressEvent) => {
                  // 更新进度（包含当前分片的部分进度）
                  const chunkProgress =
                    (progressEvent.loaded / progressEvent.total) *
                    chunk.blob.size;
                  const totalUploaded = this.progress.uploaded + chunkProgress;
                  this.updateProgress(totalUploaded);
                },
              }
            );

            if (response.data.success) {
              this.updateChunkStatus(index, "uploaded");
              // 更新总进度（加上整个分片的大小）
              const newUploaded = this.progress.uploaded + chunk.blob.size;
              this.updateProgress(newUploaded);
            } else {
              throw new Error("分片上传失败");
            }
          } catch (error) {
            this.updateChunkStatus(index, "error");
            throw error;
          }
        }

        async mergeFile() {
          try {
            const response = await axios.get(`${CONFIG.SERVER_URL}/merge`, {
              params: {
                name: `${this.randomStr}_${this.file.name}`,
              },
            });

            if (response.data.success) {
              return response.data;
            } else {
              throw new Error("文件合并失败");
            }
          } catch (error) {
            throw error;
          }
        }

        cancelUpload() {
          if (this.uploading) {
            this.cancelled = true;
            this.uploading = false;
            this.elements.uploadBtn.disabled = false;
            this.elements.uploadBtn.textContent = "开始上传";
            this.showStatus("正在取消上传...", "info");
          }
        }

        showStatus(message, type = "info") {
          this.elements.statusMessage.textContent = message;
          this.elements.statusMessage.className = "status-message";

          if (type === "success") {
```

```typescript
// 检查已上传分片
async checkUploadedChunks(fileName: string) {
  const [randomStr, originalName] = fileName.split('_', 2);
  const chunkDir = path.join(this.uploadDir, `chunks_${randomStr}_${originalName}`);

  if (!fs.existsSync(chunkDir)) {
    return { uploadedChunks: [], totalChunks: 0 };
  }

  const chunks = fs.readdirSync(chunkDir)
    .filter(file => file.startsWith('chunk_'))
    .map(file => parseInt(file.split('_')[1]));

  return {
    uploadedChunks: chunks,
    totalChunks: Math.ceil(this.getFileSize(fileName) / CHUNK_SIZE)
  };
}
```

```typescript
// 计算分片MD5，确保完整性
async calculateChunkHash(chunk: Buffer): Promise<string> {
  const crypto = require('crypto');
  return crypto.createHash('md5').update(chunk).digest('hex');
}

// 上传时校验分片
async verifyChunk(chunkIndex: number, expectedHash: string): Promise<boolean> {
  const chunkPath = this.getChunkPath(chunkIndex);
  const chunkBuffer = await fs.promises.readFile(chunkPath);
  const actualHash = await this.calculateChunkHash(chunkBuffer);
  return actualHash === expectedHash;
}
```

### 4.3 并发控制

```javascript
// 客户端并发控制
class ConcurrentUploader {
  constructor(maxConcurrent = 3) {
    this.maxConcurrent = maxConcurrent;
    this.queue = [];
    this.active = new Set();
  }

  async add(task) {
    return new Promise((resolve, reject) => {
      this.queue.push({ task, resolve, reject });
      this.next();
    });
  }

  async next() {
    if (this.active.size >= this.maxConcurrent || this.queue.length === 0) {
      return;
    }

    const { task, resolve, reject } = this.queue.shift();
    const promise = task();
    this.active.add(promise);

    promise
      .then(resolve)
      .catch(reject)
      .finally(() => {
        this.active.delete(promise);
        this.next();
      });
  }
}
```
