## 1. 概述

项目最终有两张子图：`story_planning` 负责从需求生成分镜，`prepare_storyboard_review` 负责质检、自动修订与送审。两张子图都通过普通节点接入父图，并在返回时对 `events` 做增量切片。

## 2. 分镜子图：story_planning

**1. 移动节点**

将目前所有节点移动之 `nodes/story_planning` 目录下。

更新 `nodes` 目录下的 `__init__.py` 文件：

**2. 将原来的图变成一张子图**

在 `workflows` 下新建 `story_planning.py` ，然后将之前 `graph.py` 里面的所有内容原封不动的放入 `story_planning.py`

**3. 创建 graph.py**

重新创建 `graph.py`，代码如下：

```py
from langgraph.graph import END, START, StateGraph

from server.model import TextModel, build_text_model
from server.states import ComicDramaState
from server.workflows.story_planning import build_story_planning_graph


def build_graph(text_model: TextModel):
    planning_graph = build_story_planning_graph(text_model)

    def story_planning(state: ComicDramaState) -> dict[str, object]:
        child_result = planning_graph.invoke(state)
        # 子图返回完整事件历史；父图 reducer 只追加本轮新事件。
        return {
            **child_result,
            "events": child_result["events"][len(state["events"]) :],
        }

    builder = StateGraph(ComicDramaState)

    # 添加节点
    builder.add_node("story_planning", story_planning)

    # 连边
    builder.add_edge(START, "story_planning")
    builder.add_edge("story_planning", END)

    return builder.compile()


graph = build_graph(build_text_model())
```

**注意点1**

`graph.py` 中引入 `build_story_planning_graph` 方法时，**<u>必须要写绝对路径</u>**，如下：

`from server.workflows.story_planning import build_story_planning_graph`

不能写相对路径：`from .story_planning import build_story_planning_graph`

原因：

LangGraph 会把 `graph.py` 当独立脚本执行，这时它没有父包（`__package__` 为空）。

`.` 表示“当前包”，**<u>没有父包就不能用相对导入</u>**，于是会报错：

```
ImportError: attempted relative import with no known parent package
```

**注意点2**

```py
return {
    **child_result,
    "events": child_result["events"][len(state["events"]) :],
}
```

这两行是在把**<u>子图的完整结果交回父图</u>**，`planning_graph.invoke(state)` 跑完子图后，`child_result` 是**整份最终 State**，里面既有旧字段，也有本轮新写的字段。父图节点必须返回一份 **更新 dict**，不是整份状态对象。

**`**child_result`\*\*

把**<u>子图结果拆开，原样写进返回值</u>**。`requirement`、`story_outline`、`characters`、`current_stage` 这些字段在父图里是**<u>覆盖</u>**写入，所以整份带回去没问题。

**`"events": child_result["events"][len(state["events"]) :]`**

`events` 不一样。它在 State 里声明了 `operator.add`：

```python
    events: Annotated[list[str], operator.add]
```

如果原样交回去：

- 进入子图前：`["已接收用户创意"]`
- 子图结束后：`["已接收用户创意", "已生成结构化创意需求", "已生成故事大纲", ...]`
- 父图再 `add` 一次：历史会出现两遍
  - `["已接收用户创意", "已接收用户创意", "已生成结构化创意需求", "已生成故事大纲", ...]`

所以这里只切出**本轮新增**的那一段：从进入子图时的长度切到末尾。上面的例子里，`len(state["events"])` 是 1，切片结果是：

```python
["已生成结构化创意需求", "已生成故事大纲"]
```

父图再 `add`，历史就只追加一次。

后面的 `"events"` 会覆盖前面 `**child_result` 里的同名字段，所以顺序必须是：先展开全部，再单独改 `events`。

### 3.1 创建子图

`prepare_storyboard_review.py`

```py
# 一张子图，负责把当前分镜送到可以人工审核的状态

from typing import Literal

from langgraph.graph import END, START, StateGraph

from server.model import TextModel
from server.nodes import (
    hard_rules_failed,
    make_evaluate_storyboard_quality,
    make_revise_storyboard,
    mark_quality_failed,
    mark_quality_passed,
    revision_exhausted,
    revision_stalled,
    route_after_failure,
    route_after_revision,
    route_quality,
    validate_storyboard,
)
from server.states import ComicDramaState


def route_prepare_entry(
    state: ComicDramaState,
) -> Literal[
    "validate_revised_storyboard",
    "evaluate_storyboard_quality",
    "hard_rules_failed",
]:
    validation = state.get("storyboard_validation")
    if validation is None:
        return "validate_revised_storyboard"
    if validation.is_valid:
        return "evaluate_storyboard_quality"
    return "hard_rules_failed"


def route_validated_storyboard(
    state: ComicDramaState,
) -> Literal["evaluate_storyboard_quality", "hard_rules_failed"]:
    return (
        "evaluate_storyboard_quality"
        if state["storyboard_validation"].is_valid
        else "hard_rules_failed"
    )


def build_prepare_storyboard_review_graph(text_model: TextModel):
    builder = StateGraph(ComicDramaState)

    builder.add_node("hard_rules_failed", hard_rules_failed)
    builder.add_node(
        "evaluate_storyboard_quality", make_evaluate_storyboard_quality(text_model)
    )
    builder.add_node("quality_passed", mark_quality_passed)
    builder.add_node("quality_failed", mark_quality_failed)
    builder.add_node("revise_storyboard", make_revise_storyboard(text_model))
    builder.add_node("validate_revised_storyboard", validate_storyboard)
    builder.add_node("revision_exhausted", revision_exhausted)
    builder.add_node("revision_stalled", revision_stalled)

    builder.add_conditional_edges(START, route_prepare_entry)
    builder.add_conditional_edges("hard_rules_failed", route_after_failure)
    builder.add_conditional_edges("evaluate_storyboard_quality", route_quality)
    builder.add_edge("quality_passed", END)
    builder.add_conditional_edges("quality_failed", route_after_failure)
    builder.add_conditional_edges("revise_storyboard", route_after_revision)
    builder.add_conditional_edges(
        "validate_revised_storyboard", route_validated_storyboard
    )
    builder.add_edge("revision_exhausted", END)
    builder.add_edge("revision_stalled", END)

    return builder.compile()

```

### 3.2 父图中接入这张子图

`graph.py`

```py
# 主图（父图）

from langgraph.graph import END, START, StateGraph

from server.model import TextModel, build_text_model
from server.nodes import (
    initialize_revision,
    make_revise_from_review,
    review_storyboard,
    storyboard_approved,
    storyboard_cancelled,
)
from server.states import ComicDramaState
from server.workflows.prepare_storyboard_review import (
    build_prepare_storyboard_review_graph,
)
from server.workflows.story_planning import build_story_planning_graph


def build_graph(text_model: TextModel, *, checkpointer=None):
    # 创建子图
    planning_graph = build_story_planning_graph(text_model)
    prepare_graph = build_prepare_storyboard_review_graph(text_model)

    # 新增一个节点
    def story_planning(state: ComicDramaState) -> dict[str, object]:
        # 调用子图
        child_result = planning_graph.invoke(state)

        return {
            **child_result,
            "events": child_result["events"][len(state["events"]) :],
        }

    def prepare_storyboard_review(state: ComicDramaState) -> dict[str, object]:
        # 调用子图
        child_result = prepare_graph.invoke(state)

        return {
            **child_result,
            "events": child_result["events"][len(state["events"]) :],
        }

    builder = StateGraph(ComicDramaState)

    # 加点
    builder.add_node("story_planning", story_planning)
    builder.add_node("initialize_revision", initialize_revision)
    builder.add_node("prepare_storyboard_review", prepare_storyboard_review)
    builder.add_node("review_storyboard", review_storyboard)
    builder.add_node("revise_from_review", make_revise_from_review(text_model))
    builder.add_node("storyboard_approved", storyboard_approved)
    builder.add_node("storyboard_cancelled", storyboard_cancelled)

    # 连边
    builder.add_edge(START, "story_planning")
    builder.add_edge("story_planning", "initialize_revision")
    builder.add_edge("initialize_revision", "prepare_storyboard_review")
    builder.add_edge("prepare_storyboard_review", "review_storyboard")

    # review_storyboard 由 Command 选择唯一业务路径，不再添加静态出口。
    builder.add_conditional_edges(
        "revise_from_review",
        lambda state: (
            "review_storyboard"
            if state["current_stage"] == "revision_stalled"
            else "prepare_storyboard_review"
        ),
        path_map=["review_storyboard", "prepare_storyboard_review"],
    )
    builder.add_edge("storyboard_approved", END)
    builder.add_edge("storyboard_cancelled", END)
    return builder.compile(checkpointer=checkpointer)


graph = build_graph(build_text_model())

```
