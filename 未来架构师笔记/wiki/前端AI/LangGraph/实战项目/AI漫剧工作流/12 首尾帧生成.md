## 1. 生成首尾帧

### 1.1 工作流

<img src="https://picgo-1300696809.cos.ap-beijing.myqcloud.com/obsidian/1794730188301_2026-11-05-165958.png" alt="image-20261106005957811" style="zoom:50%;" />

（PPT 1）

这里工作流新增的部分，总结起来就3件事：

1. 先生成一套分镜的首帧和尾帧
2. 中断，让用户检查和审核
3. 有问题就只重做对应帧，直到批准或取消。

图中的几个分支可以这样理解：

- **`check_frames → prepare_frame_redos`**：程序发现缺图、坏图等问题，而且还有重做额度，自动补图。
- **`check_frames → review_frames`**：检查通过，交给用户判断画面效果；失败帧额度耗尽时，也会暂停让用户查看，但不能批准。
- **`review_frames → prepare_frame_redos`**：图片文件没问题，但用户觉得人物、构图或动作不满意，要求**<u>局部</u>**重做。
- **`review_frames → review_frames`**：提交的 JSON、版本或令牌有误，提示错误后重新等用户填写。

`frame_review.py`

```py
"""画面审核决定和精确到单帧的返工意见。"""

from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, model_validator

FrameType = Literal["start", "end"]
FrameReviewAction = Literal["approve", "regenerate", "cancel"]


class FrameRedoItem(BaseModel):
    """精确到单帧的返工意见。reason 只记录原因，comment 才会追加到图片提示词"""
    model_config = ConfigDict(extra="forbid", str_strip_whitespace=True)
    shot_id: str = Field(pattern=r"^shot_[a-z0-9_]+$")
    frame_type: FrameType
    reason: str = Field(min_length=1, max_length=300)
    comment: str = Field(default="", max_length=500)


class FrameReview(BaseModel):
    """一轮画面审核决定。version 和 review_token 必须对应当前审核快照"""
    model_config = ConfigDict(extra="forbid", str_strip_whitespace=True)
    command_id: str | None = Field(default=None, pattern=r"^[a-z0-9_]{1,80}$")
    action: FrameReviewAction
    version: int = Field(ge=0)
    review_token: str = Field(pattern=r"^[a-f0-9]{64}$")
    items: list[FrameRedoItem] = Field(default_factory=list, max_length=12)

    @model_validator(mode="after")
    def check_items(self):
        if self.action == "regenerate" and not self.items:
            raise ValueError("重做必须指定至少一帧")
        if self.action != "regenerate" and self.items:
            raise ValueError("通过和取消不能携带重做项")
        if self.command_id is None:
            items = sorted(
                {
                    (item.shot_id, item.frame_type, item.reason, item.comment)
                    for item in self.items
                }
            )
            payload = {
                "review_token": self.review_token,
                "version": self.version,
                "action": self.action,
                "items": items,
            }
            encoded = json.dumps(
                payload, sort_keys=True, ensure_ascii=False, separators=(",", ":")
            ).encode("utf-8")
            self.command_id = "review_" + hashlib.sha256(encoded).hexdigest()
        return self
```

重点说一下 `review_token`：这是**<u>当前这一轮待审核画面的指纹</u>**，用来确认用户提交的决定。（PPT 11）

然后再说一下 `command_id`，这是提交审核决定的一方（用户）自定义的命令编号，用来识别“这次决定是否已经处理过”。

例如：

```json
"command_id": "approve_frames_01"
"command_id": "redo_frames_01"
"command_id": "redo_frames_02"
"command_id": "cancel_frames_01"
```

如果同一条重做命令被重复提交：

```json
command_id = redo_01 → 首次接受，安排重做
command_id = redo_01 → 再次提交，被拒绝，避免重复处理
```

这个 `command_id` 可以用户自己起名，只要每次新决定使用新的编号即可。

如果用户没传，则会自动生成。

```py
"""从冻结镜头任务构造首尾帧请求，不重新生成故事文本。"""

import hashlib
from typing import Literal

from server.schema import ImageGenerationRequest, ShotFrameTask


def frame_request(task: ShotFrameTask, frame_type: Literal["start", "end"]):
    shot = task.shot
    purpose = "shot_start" if frame_type == "start" else "shot_end"
    frame_prompt = (
        shot.start_frame_prompt if frame_type == "start" else shot.end_frame_prompt
    )
    descriptions = []
    for number, character in enumerate(task.characters, start=1):
        descriptions.append(
            f"参考图{number}对应角色{character.character_id}（{character.name}）；"
            f"稳定外观：{character.appearance}；固定服装：{character.clothing}；"
            f"辨识特征：{'、'.join(character.visual_keywords)}。"
        )
    scene = task.scene
    descriptions.append(
        f"参考图{len(task.characters) + 1}对应场景{scene.scene_id}（{scene.name}）；"
        f"布局：{scene.location}；时间：{scene.time}；光线：{scene.lighting}；"
        f"色彩：{scene.colors}；环境元素：{'、'.join(scene.visual_keywords)}。"
    )
    prompt = (
        f"生成镜头{shot.shot_id}的{'首帧' if frame_type == 'start' else '尾帧'}，只生成一张静态漫画画面。\n"
        f"画风：{task.visual_style}；统一基础色彩：{task.palette}；画幅：{task.aspect_ratio}。\n"
        + "\n".join(descriptions)
        + f"\n镜头动作背景：{shot.action}；主体构图：{shot.composition}；机位/运动背景：{shot.camera_motion}。\n"
        + f"这一帧的瞬间：{frame_prompt}\n"
        + "参考图用于保持人物外观、服装与场景元素；按本帧动作重新构图，不照搬参考图的站姿或空镜。"
        + "首尾帧属于同一镜头，保持人物身份、服装和空间关系；不绘制对白、字幕或水印。"

    digest = hashlib.sha256(task.model_dump_json().encode()).hexdigest()[:16]

        task.aspect_ratio

    return ImageGenerationRequest(
        request_id=f"img_{task.project_id}_{purpose}_{shot.shot_id}_v{task.version}_{digest}",
        purpose=purpose,
        owner_id=shot.shot_id,
        version=task.version,
        prompt=prompt,
        negative_prompt=task.negative_prompt,
        size=size,
        reference_images=task.reference_images,
```

```py
def reference_from_image(self, image: GeneratedImage) -> ImageReference:
    path = (self.project_root / image.local_path).resolve()
    content = self._read_reference_file(path)
    return ImageReference(
        local_path=path.relative_to(self.project_root).as_posix(),
        content_sha256=hashlib.sha256(content).hexdigest(),
    )

def read_reference(self, reference: ImageReference) -> bytes:
    path = (self.project_root / reference.local_path).resolve()
    content = self._read_reference_file(path)
    if hashlib.sha256(content).hexdigest() != reference.content_sha256:
        raise ValueError("参考图内容已变化，不能继续使用冻结任务")
    return content

def _read_reference_file(self, path: Path) -> bytes:
    if not path.is_relative_to(self.output_root):
        raise ValueError("参考图必须位于当前项目 outputs 中")
    with path.open("rb") as file:
        content = file.read(20 * 1024 * 1024 + 1)
    if len(content) > 20 * 1024 * 1024 or not self._looks_like_image(content):
        raise ValueError("参考图过大或格式无效")
    return content
```

### 1.6 services

在进入 services 之前，先看环境变量的变化：

```bash
# 首尾帧阶段的最大并发
SHOT_FRAME_MAX_CONCURRENCY=3
# 配合 FAKE_IMAGE_FAIL_OWNER_IDS 精确选择失败帧，如 shot_end；空值表示不限用途
FAKE_IMAGE_FAIL_PURPOSES=
```

`FAKE_IMAGE_FAIL_PURPOSES` 用来限定假图生成器让哪一种图片失败，空着表示不按用途筛选。

它必须和 `FAKE_IMAGE_FAILURE` 一起用。后者设成 `rate_limit`、`content_rejected` 或 `service_error` 时，生成器才会模拟失败；设成 `none` 时，这一项写什么都不会失败。

用途对应请求里的 `purpose`，可以写一个或多个，用逗号分开：

- `character_reference`：角色参考图
- `scene_reference`：场景参考图
- `shot_start`：镜头首帧
- `shot_end`：镜头尾帧

例如 `FAKE_IMAGE_FAILURE=service_error` 且 `FAKE_IMAGE_FAIL_PURPOSES=shot_end`，只有尾帧会失败，首帧和参考图仍保存那张固定假图。再配上 `FAKE_IMAGE_FAIL_OWNER_IDS=shot_03`，就只有第三镜的尾帧失败。

这段代码是**检查本次要使用的参考图片是否还有效**。

这里的 `reference_images` 是生成当前画面所依据的角色、场景参考图。每项记录了图片路径，以及准备任务时计算的文件内容摘要。

`read_reference()` 会读取文件并检查：

- 路径是否在项目的素材目录内。
- 文件是否存在，大小和格式是否符合基本要求。
- **当前文件内容是否与任务准备时一致**。

例如，准备任务时使用的是“阿沐穿白衣”的参考图，之后却把同一路径的文件换成了“阿沐穿黑衣”的图。路径没变，但内容摘要变了，这里就会报错，避免同一份冻结任务悄悄换了生成依据。

因此，**<u>前者检查输入是否有效，后者检查输出能否复用</u>**。

Pillow 是 Python 里最常用的图片处理库，它是老库 PIL（Python Imaging Library）的继任者。

它能打开、解码、缩放、裁剪、转格式、画图，也支持 PNG、JPEG、WEBP 等常见格式。在这个项目里面，它用于：

1. 校验已生成的首尾帧素材
2. 读入图片字节
3. 确认是 PNG / JPEG / WEBP
4. 读出宽高
5. 拦住过大的图（超过 1600 万像素，触发 `Image.DecompressionBombError`）

整个 `helpers.py` 代码如下：

```py
"""共用工具：帧标识、审核快照、新版本请求与图片检查。"""

import hashlib
import json
from io import BytesIO

from PIL import Image

from server.prompts.frame_prompts import frame_request
from server.schema import (
    Asset,
    FrameRedoItem,
    ImageGenerationRequest,
    ImageReference,
    ShotFrameTask,
)
from server.schema.frame_review import FrameType
from server.storage import AssetStore


def frame_key(shot_id: str, frame_type: FrameType) -> str:
    """拼出一帧的逻辑键，用来在索引里选中当前这一帧。

    入参：shot_id 是镜头 ID，如 shot_03；frame_type 是 start（首帧）或 end（尾帧）。
    返回：形如 shot_03:end 的字符串。它只用来定位当前帧，和保存每次生成档案的 request_id 不是同一个标识。
    """
    return f"{shot_id}:{frame_type}"


def snapshot_token(
    project_id: str, version: int, round_number: int, frames: list
) -> str:
    """为当前这一轮画面计算审核令牌，提交审核时必须带上同一个值。

    入参：project_id 是作品 ID；version 是已批准的分镜版本；round_number 是审核轮次；
    frames 是这一轮各帧的检查记录，其中包含文件内容摘要。
    返回：64 位十六进制字符串。重做后即使分镜文字没变，轮次或文件摘要变了也会得到新令牌。
    """
    payload = [project_id, version, round_number, frames]
    encoded = json.dumps(payload, sort_keys=True, ensure_ascii=False).encode()
    return hashlib.sha256(encoded).hexdigest()


def redo_request(previous: ImageGenerationRequest, item: FrameRedoItem):
    """根据当前候选请求和返工意见，造出下一版生图请求。

    入参：previous 是这一帧当前的 ImageGenerationRequest；item 是一条 FrameRedoItem。
    返回：新的 ImageGenerationRequest。version 加 1；item.comment 非空时追加到提示词末尾；
    request_id 按新内容重新计算。item.reason 只记录原因，不进入提示词。
    """
    data = previous.model_dump(mode="json")
    data["version"] += 1
    if item.comment:
        data["prompt"] += f"\n本次画面修改要求：{item.comment}"
    # 【重要】稳定新 ID 绑定旧请求、素材版本与意见。重放同一计划时可以命中缓存；
    # 换意见生成新 ID，旧图片和旧元数据不会被原子写入静默覆盖。
    digest = hashlib.sha256(json.dumps(data, sort_keys=True).encode()).hexdigest()[:16]
    data["request_id"] = (
        f"img_redo_{previous.owner_id}_{previous.purpose}_v{data['version']}_{digest}"
    )
    return ImageGenerationRequest.model_validate(data)


MAX_FRAME_REDOS = 2


def load_reference(
    store: AssetStore, assets: dict, request: ImageGenerationRequest
) -> ImageReference:
    """核对参考图请求、State 记录和磁盘文件，返回冻结了实际内容的参考图。

    入参：store 是素材库；assets 是状态里的素材表，键为 request_id；
    request 是角色或场景参考图的生图请求。
    返回：ImageReference，含相对路径和文件内容摘要。记录缺失、失败、与请求不一致，
    或磁盘档案对不上时抛出 ValueError。
    """
    payload = assets.get(request.request_id)
    asset = Asset.model_validate(payload) if payload else None
    # 【重要】先核对 State 与当前请求，再重新验证磁盘文件。
    # reference_report.is_complete 不能证明文件现在仍然存在。
    expected = request.model_dump(mode="json")
    if (
        asset is None
        or asset.status != "succeeded"
        or any(
            asset.model_dump(mode="json")[name] != value
            for name, value in expected.items()
        )
    ):
        raise ValueError("参考素材缺失、失败或与当前请求不一致")
    image = store.load(request)
    if image is None or asset.image.local_path != image.local_path:
        raise ValueError("参考素材没有匹配的落盘档案")
    return store.reference_from_image(image)


def frame_project_id(state):
    """取出并核对当前作品 ID。

    入参：state 是图状态，必须包含 visual_bible.project_id；state.project_id 可以不填。
    返回：冻结视觉设定里的作品 ID。状态里另外写了 project_id 且两边不一致时抛出 ValueError。
    """
    project_id = state["visual_bible"]["project_id"]
    if state.get("project_id", project_id) != project_id:
        raise ValueError("作品 ID 与冻结视觉设定不一致")
    return project_id


def initial_frame_index(state):
    """按冻结的镜头任务，重建每一帧第一次生成时的生图请求。

    入参：state 是图状态，使用 approved_version、revision_count、frame_tasks 和 visual_bible。
    返回：字典。键是 frame_key，值是 ImageGenerationRequest 的字典。这里只有首版请求，不含重做后的候选。
    """
    if state["approved_version"] != state["revision_count"]:
        raise ValueError("分镜版本发生变化，不能审核旧帧")
    index = {}
    tasks = [ShotFrameTask.model_validate(x) for x in state["frame_tasks"]]
    if not 4 <= len(tasks) <= 6 or len({x.shot.shot_id for x in tasks}) != len(tasks):
        raise ValueError("冻结镜头任务数量或 ID 不合法")
    for task in sorted(tasks, key=lambda x: x.shot.order):
        if (
            task.project_id != frame_project_id(state)
            or task.version != state["approved_version"]
        ):
            raise ValueError("帧任务不属于当前批准作品版本")
        for kind in ("start", "end"):
            index[frame_key(task.shot.shot_id, kind)] = frame_request(
                task, kind
            ).model_dump(mode="json")
    return index


def inspect_frames(state, store: AssetStore):
    """重新打开每一帧的实际文件做检查，不沿用 frame_report 里的成功标记。

    入参：state 是图状态；store 是素材库，用来读取参考图和已经落盘的帧。
    返回：(index, frames, failed)。
    index 是当前候选帧索引，键为 frame_key，值为 ImageGenerationRequest 字典。
    frames 是每一帧的检查记录：成功的含路径、宽高和内容摘要，失败的 local_path 为 None。
    failed 是失败帧列表，每一项是 FrameRedoItem 的字典。
    """
    baseline = initial_frame_index(state)
    index = state.get("frame_index", baseline)
    if set(index) != set(baseline):
        raise ValueError("当前帧索引必须完整对应冻结镜头")
    frames, failed = [], []
    for key in baseline:  # 冻结顺序仅用于展示；查找始终使用逻辑键。
        request = ImageGenerationRequest.model_validate(index[key])
        shot_id, kind = key.split(":")
        if request.owner_id != shot_id or request.purpose != f"shot_{kind}":
            raise ValueError("帧索引与请求归属不一致")
        original = ImageGenerationRequest.model_validate(baseline[key])
        if (
            any(
                getattr(request, name) != getattr(original, name)
                for name in ("reference_images", "size", "negative_prompt")
            )
            or request.version < original.version
        ):
            raise ValueError("候选帧不能改变冻结参考、尺寸或回退版本")
        entry = {
            "key": key,
            "shot_id": shot_id,
            "frame_type": kind,
            "version": request.version,
            "request_id": request.request_id,
        }
        # 最近一次重做的旧版本供人工对照；只展示档案，不把旧图认作当前合格候选。
        for task in state.get("frame_redo_tasks", []):
            if (
                task["key"] == key
                and task["request"]["request_id"] == request.request_id
            ):
                old = state.get("assets", {}).get(task["previous_request_id"], {})
                entry["previous"] = {
                    "request_id": task["previous_request_id"],
                    "version": old.get("version"),
                    "local_path": (old.get("image") or {}).get("local_path"),
                }
        error = "missing_asset"
        try:
            raw = state.get("assets", {}).get(request.request_id)
            asset = Asset.model_validate(raw) if raw else None
            if asset:
                expected = request.model_dump(mode="json")
                if any(
                    asset.model_dump(mode="json")[k] != v for k, v in expected.items()
                ):
                    raise ValueError("素材请求不一致")
                error = (
                    asset.error_code
                    if asset.status == "failed"
                    else "asset_storage_invalid"
                )
                if asset.status == "succeeded":
                    for source in request.reference_images:
                        store.read_reference(source)
                    loaded = store.load(request)
                    if loaded is None or loaded.local_path != asset.image.local_path:
                        raise ValueError("素材没有有效档案")
                    reference = store.reference_from_image(loaded)
                    content = store.read_reference(reference)
                    # open() 只识别文件，load() 才执行解码；限制像素后再分配解码内存。
                    with Image.open(
                        BytesIO(content), formats=["PNG", "JPEG", "WEBP"]
                    ) as image:
                        width, height = image.size
                        if width * height > 16_000_000:
                            raise ValueError("素材超过 1600 万像素上限")
                        image.load()
                        # Fake 明确只验编排用的 1×1；真实图严格匹配本次请求的数字尺寸。
                        if loaded.provider_model == "fake-image-v1":
                            if (width, height) != (1, 1):
                                raise ValueError("Fake 图片尺寸不符合约定")
                        elif "x" not in request.size or (width, height) != tuple(
                            map(int, request.size.split("x"))
                        ):
                            raise ValueError(
                                "实际尺寸不符合请求；不能用 size 元数据冒充解码结果"
                            )
                    frames.append(
                        {
                            **entry,
                            "local_path": loaded.local_path,
                            "content_sha256": reference.content_sha256,
                            "width": width,
                            "height": height,
                            "error_code": None,
                        }
                    )
                    continue
        except (OSError, ValueError, Image.DecompressionBombError):
            error = "asset_storage_invalid"
        failed.append(
            FrameRedoItem(shot_id=shot_id, frame_type=kind, reason=error).model_dump(
                mode="json"
            )
        )
        frames.append({**entry, "local_path": None, "error_code": error})
    return index, frames, failed

```

#### 1.7.2 7-2.`generation.py`

主要负责第一次批量生成所有分镜头的首尾帧：

```py
"""首次首尾帧制作：准备镜头任务、动态派发、生成与汇总。"""

from typing import TypedDict

from langgraph.types import Send

from server.nodes.frames.helpers import load_reference
from server.nodes.references.visual_bible import _bible
from server.prompts.frame_prompts import frame_request
from server.prompts.reference_prompts import (
    character_reference_request,
    scene_reference_request,
    visual_digest,
)
from server.schema import (
    Asset,
    CreativeRequirement,
    ShotFrameTask,
    StoryBoard,
    VisualBible,
)
from server.services.image import ImageGenerationError, ImageGenerator
from server.states import ComicDramaState
from server.storage import AssetStore


class ShotFrameInput(TypedDict):
    # Send 的子任务输入只有 task，不是整个 ComicDramaState。
    task: dict


def route_after_references(state: ComicDramaState):
    return (
        "prepare_shot_frames"
        if state["current_stage"] == "reference_assets_ready"
        else "__end__"
    )


def make_prepare_shot_frames(store: AssetStore, generation_profile: str):
    def prepare_shot_frames(state: ComicDramaState) -> dict:
        if state["current_stage"] != "reference_assets_ready":
            raise ValueError("参考素材尚未就绪，不能派发首尾帧")
        bible = VisualBible.model_validate(state["visual_bible"])
        # 复用已有视觉快照校验，连同当前配置再次核对。
        if (
            bible != _bible(state, generation_profile)
            or visual_digest(bible) != state["visual_digest"]
        ):
            raise ValueError("审核设定、视觉快照或生成配置已变化")
        shots = [StoryBoard.model_validate(x) for x in state["storyboard"]]
        ids = [x.shot_id for x in shots]
        orders = [x.order for x in shots]
        if (
            not 4 <= len(shots) <= 6
            or len(set(ids)) != len(ids)
            or sorted(orders) != list(range(1, len(shots) + 1))
        ):
            raise ValueError("镜头数量必须为 4～6，ID 唯一，order 连续")
        if (
            len(shots)
            != CreativeRequirement.model_validate(state["requirement"]).shot_count
        ):
            raise ValueError("镜头数量与审核需求不一致")
        characters = {x.character_id: x for x in bible.characters}
        scenes = {x.scene_id: x for x in bible.scenes}
        assets = state.get("assets", {})

        tasks = []
        for shot in sorted(shots, key=lambda x: x.order):
            if (
                shot.scene_id not in scenes
                or len(set(shot.character_ids)) != len(shot.character_ids)
                or not set(shot.character_ids) <= characters.keys()
            ):
                raise ValueError("分镜引用了未知或重复角色/未知场景")
            related = [characters[key] for key in shot.character_ids]
            scene = scenes[shot.scene_id]
            inputs = [
                load_reference(store, assets, character_reference_request(bible, x))
                for x in related
            ]
            inputs.append(
                load_reference(store, assets, scene_reference_request(bible, scene))
            )
            if len(inputs) > 8:
                raise ValueError("单镜头参考图超过上限 8 张")
            task = ShotFrameTask(
                project_id=bible.project_id,
                version=bible.version,
                visual_digest=state["visual_digest"],
                generation_profile=generation_profile,
                visual_style=bible.visual_style,
                palette=bible.palette,
                aspect_ratio=bible.aspect_ratio,
                negative_prompt=bible.negative_prompt,
                shot=shot,
                characters=related,
                scene=scene,
                reference_images=inputs,
            )
            # 在派发前构造两请求，确保提示词长度、ID 等 Schema 错误不会触发部分付费调用。
            frame_request(task, "start")
            frame_request(task, "end")
            tasks.append(task.model_dump(mode="json"))
        if state.get("frame_tasks") is not None and state["frame_tasks"] != tasks:
            raise ValueError("冻结镜头任务已变化，不能沿用旧任务更换镜头或参考图")
        return {
            "frame_tasks": tasks,
            "frame_report": {},
            "current_stage": "shot_frames_generating",
            "events": [f"已准备 {len(tasks)} 个首尾帧镜头任务"],
        }

    return prepare_shot_frames


def dispatch_shot_frames(state: ComicDramaState) -> list[Send]:
    tasks = state.get("frame_tasks", [])
    if state["current_stage"] != "shot_frames_generating" or not tasks:
        raise ValueError("必须先准备非空镜头任务")
    # 这里只有一个注册节点名，运行时为每个镜头生成一个任务实例。
    # task 是 JSON 基础类型，客户端通过 make_generate_shot_frames 的闭包注入。
    return [Send("generate_shot_frames", {"task": task}) for task in tasks]


def make_generate_shot_frames(generator: ImageGenerator):
    def generate_shot_frames(state: ShotFrameInput) -> dict:
        task = ShotFrameTask.model_validate(state["task"])
        updates = {}
        for frame_type in ("start", "end"):
            request = frame_request(task, frame_type)
            common = request.model_dump(mode="json")
            try:
                # generator 先核验参考图和已落盘帧；命中时返回 reused=True。
                # 两帧在同一镜头任务里依次执行，不另开线程，因此一个任务最多一笔生成在途。
                image = generator.generate(request)
                asset = Asset(**common, status="succeeded", image=image)
            except ImageGenerationError as exc:
                # 首帧成功、尾帧失败时，两项都返回；其他镜头不会被这项业务错误撤销。
                asset = Asset(**common, status="failed", error_code=exc.code)
            except (OSError, ValueError):
                # 文件损坏、参考图变化或适配结果不一致，记录稳定错误，不自动重复付费。
                asset = Asset(
                    **common, status="failed", error_code="asset_storage_invalid"
                )
            updates[request.request_id] = asset.model_dump(mode="json")

        return {"assets": updates, "events": [f"{task.shot.shot_id} 首尾帧任务完成"]}

    return generate_shot_frames


def make_collect_shot_frames(store: AssetStore):
    def collect_shot_frames(state: ComicDramaState) -> dict:
        succeeded, failed, ordered = [], [], []
        tasks = [ShotFrameTask.model_validate(x) for x in state["frame_tasks"]]
        for task in sorted(tasks, key=lambda x: x.shot.order):
            for frame_type in ("start", "end"):
                request = frame_request(task, frame_type)
                payload = state.get("assets", {}).get(request.request_id)
                error, image = "missing_asset", None
                try:
                    asset = Asset.model_validate(payload) if payload else None
                    if asset is not None:
                        expected = request.model_dump(mode="json")
                        if any(
                            asset.model_dump(mode="json")[key] != value
                            for key, value in expected.items()
                        ):
                            error = "asset_request_mismatch"
                        elif asset.status == "failed":
                            error = asset.error_code
                        else:
                            # 再确认实际档案；reused 字段仍保留本次生成节点返回的值。
                            loaded = store.load(request)
                            if (
                                loaded is None
                                or loaded.local_path != asset.image.local_path
                            ):
                                error = "asset_storage_invalid"
                            else:
                                image = asset.image
                except (OSError, ValueError):
                    error = "asset_storage_invalid"
                entry = {
                    "shot_id": task.shot.shot_id,
                    "frame_type": frame_type,
                    "version": task.version,
                    "request_id": request.request_id,
                }
                if image is None:
                    failed.append({**entry, "error_code": error})
                else:
                    succeeded.append(request.request_id)
                    ordered.append(
                        {
                            **entry,
                            "local_path": image.local_path,
                            "reused": image.reused,
                        }
                    )
        stage = (
            "shot_frames_ready"
            if not failed
            else ("shot_frames_partial_failed" if succeeded else "shot_frames_failed")
        )
        return {
            "frame_report": {
                "expected_count": len(tasks) * 2,
                "succeeded_ids": succeeded,
                "failed": failed,
                "ordered_frames": ordered,
                "is_complete": not failed,
            },
            "current_stage": stage,
            "events": [f"首尾帧汇总：成功 {len(succeeded)}，失败 {len(failed)}"],
        }

    return collect_shot_frames
```

阶段按失败情况分成三种：

- `failed` 是空的：全部成功，阶段是 `shot_frames_ready`。
- `failed` 有内容，同时 `succeeded` 也有内容：有的帧成了、有的帧没成，阶段是 `shot_frames_partial_failed`。
- `failed` 有内容，`succeeded` 是空的：一张都没成功，阶段是 `shot_frames_failed`。

`succeeded` 和 `ordered` 两个列表一起增长，只说明它们记的是同一批成功帧；阶段判断要分的是三种结果，所以看 `failed` 和 `succeeded` 就够了。

`ordered` 和 `succeeded` 条数相同，所以这里换成看 `ordered` 是否为空，三种阶段的结果也一样。

#### 1.7.3 7-3. `review.py`

```py
"""画面检查与人工审核：检查、决定、批准和取消。"""

from typing import Literal

from langgraph.types import Command, interrupt
from pydantic import ValidationError

from server.schema import FrameReview, ImageGenerationRequest
from server.states import ComicDramaState
from server.storage import AssetStore

from .helpers import (
    MAX_FRAME_REDOS,
    frame_key,
    frame_project_id,
    inspect_frames,
    redo_request,
    snapshot_token,
)


def make_check_frames(store: AssetStore):
    def check_frames(
        state: ComicDramaState,
    ) -> Command[Literal["prepare_frame_redos", "review_frames"]]:
        index, frames, failed = inspect_frames(state, store)
        counts = state.get("frame_redo_counts", {})
        exhausted = [
            frame_key(x["shot_id"], x["frame_type"])
            for x in failed
            if counts.get(frame_key(x["shot_id"], x["frame_type"]), 0)
            >= MAX_FRAME_REDOS
        ]
        round_number = state.get("frame_review_round", 0) + 1
        token = snapshot_token(
            frame_project_id(state), state["approved_version"], round_number, frames
        )
        # 有一项耗尽就暂停，保留其他项的额度；不把整部作品无限拖进生成循环。
        auto_redo = bool(failed) and not exhausted
        return Command(
            goto="prepare_frame_redos" if auto_redo else "review_frames",
            update={
                "frame_index": index,
                "frame_check": {
                    "frames": frames,
                    "failed": failed,
                    "is_complete": not failed,
                    "exhausted_keys": exhausted,
                },
                "pending_frame_redos": failed if auto_redo else [],
                "frame_review_round": round_number,
                "frame_review_token": token,
                "frame_review_error": None,
                "approved_frames": {},
                "current_stage": (
                    "frame_redo_pending"
                    if auto_redo
                    else (
                        "frame_redo_exhausted" if exhausted else "frames_review_pending"
                    )
                ),
                "events": [f"画面检查：失败 {len(failed)}，耗尽 {len(exhausted)}"],
            },
        )

    return check_frames


def make_review_frames(store: AssetStore):
    def review_frames(
        state: ComicDramaState,
    ) -> Command[
        Literal[
            "frames_approved",
            "prepare_frame_redos",
            "frames_cancelled",
            "review_frames",
        ]
    ]:
        # 【重要】节点恢复时从头重放。中断前只组装载荷，不生成、不写文件，
        # 每次调用仅中断一次；无效答案先保存错误，再通过图路由回本节点。
        raw = interrupt(
            {
                "project_id": frame_project_id(state),
                "version": state["approved_version"],
                "review_token": state["frame_review_token"],
                "error": state.get("frame_review_error"),
                "automatic_check": state["frame_check"],
                "allowed_actions": ["approve", "regenerate", "cancel"],
                "max_redos_per_frame": MAX_FRAME_REDOS,
                "redo_counts": state.get("frame_redo_counts", {}),
            }
        )
        try:
            review = FrameReview.model_validate(raw)
            if review.command_id in state.get("frame_command_ids", []):
                raise ValueError("该审核命令已处理，请刷新当前审核载荷后再提交")
            if (
                review.version != state["approved_version"]
                or review.review_token != state["frame_review_token"]
            ):
                raise ValueError("审核快照已过期，请使用当前 version 和 review_token")
            # 取消不要求损坏文件可恢复；通过和重做必须核对用户正在看的快照。
            if review.action != "cancel":
                _, frames, _ = inspect_frames(state, store)
                current = snapshot_token(
                    frame_project_id(state),
                    state["approved_version"],
                    state["frame_review_round"],
                    frames,
                )
                if current != review.review_token:
                    raise ValueError(
                        "审核期间文件发生变化，请取消并检查档案；不能批准旧快照"
                    )
            if review.action == "approve" and not state["frame_check"]["is_complete"]:
                raise ValueError("素材不完整或重做额度耗尽，不能批准；可取消作品")
            unique = {}
            for item in review.items:
                key = frame_key(item.shot_id, item.frame_type)
                if key not in state["frame_index"]:
                    raise ValueError("重做项引用了未知镜头或帧")
                if key in unique and unique[key] != item.model_dump(mode="json"):
                    raise ValueError("同一帧存在冲突意见，请合并后提交")
                if state.get("frame_redo_counts", {}).get(key, 0) >= MAX_FRAME_REDOS:
                    raise ValueError("该帧重做次数已达上限，可取消作品并人工检查")
                # 提前验证修改意见长度与新请求 Schema，错误在付费计划落盘前重新提问。
                redo_request(
                    ImageGenerationRequest.model_validate(state["frame_index"][key]),
                    item,
                )
                unique[key] = item.model_dump(mode="json")
        except (ValidationError, ValueError) as exc:
            # 【重要】错误进入 Checkpoint，然后 goto 自己产生新的审核任务。
            # 不在节点内 while interrupt，避免恢复时重放此前所有无效答案。
            return Command(
                goto="review_frames", update={"frame_review_error": str(exc)}
            )
        target = {
            "approve": "frames_approved",
            "regenerate": "prepare_frame_redos",
            "cancel": "frames_cancelled",
        }[review.action]
        return Command(
            goto=target,
            update={
                "frame_review_error": None,
                "frame_review_result": review.model_dump(mode="json"),
                "frame_command_ids": [
                    *state.get("frame_command_ids", []),
                    review.command_id,
                ],
                "pending_frame_redos": list(unique.values()),
                "events": [f"画面审核决定：{review.action}"],
            },
        )

    return review_frames


def make_approve_frames(store: AssetStore):
    def frames_approved(state: ComicDramaState) -> dict:
        _, frames, failed = inspect_frames(state, store)
        token = snapshot_token(
            frame_project_id(state),
            state["approved_version"],
            state["frame_review_round"],
            frames,
        )
        if (
            failed
            or token != state["frame_review_token"]
            or state["frame_review_result"]["action"] != "approve"
        ):
            raise ValueError("批准前素材再次校验失败")
        # 视频阶段只能消费这个明确锁定的清单，不能遍历历史 assets 取最大版本。
        return {
            "approved_frames": {x["key"]: x for x in frames},
            "current_stage": "frames_approved",
            "events": ["首尾帧版本已批准，等待视频阶段"],
        }

    return frames_approved

        "approved_frames": {},

        "stop_reason": "frames_cancelled_by_user",
```

```python
"""局部单帧重做：准备新版本、动态派发与生成。"""

from typing import TypedDict

from langgraph.types import Send

from server.schema import Asset, FrameRedoItem, ImageGenerationRequest
from server.services.image import ImageGenerationError, ImageGenerator
from server.states import ComicDramaState

from .helpers import frame_key, redo_request


class FrameRedoInput(TypedDict):
    redo_task: dict


def prepare_frame_redos(state: ComicDramaState) -> dict:
    index = dict(state["frame_index"])
    counts = dict(state.get("frame_redo_counts", {}))
    tasks = []
    for raw in state["pending_frame_redos"]:
        item = FrameRedoItem.model_validate(raw)
        key = frame_key(item.shot_id, item.frame_type)
        previous = ImageGenerationRequest.model_validate(index[key])
        request = redo_request(previous, item)
        tasks.append(
            {
                "key": key,
                "request": request.model_dump(mode="json"),
                "previous_request_id": previous.request_id,
            }
        )
        index[key] = request.model_dump(mode="json")
        counts[key] = counts.get(key, 0) + 1
    if not tasks:
        raise ValueError("重做计划不能为空")
    return {
        "frame_index": index,
        "frame_redo_counts": counts,
        "frame_redo_tasks": tasks,
        "pending_frame_redos": [],
        "approved_frames": {},
        "current_stage": "frames_regenerating",
        "events": [f"已准备 {len(tasks)} 个局部单帧任务"],
    }


def dispatch_frame_redos(state: ComicDramaState) -> list[Send]:
    tasks = state.get("frame_redo_tasks", [])
    if state["current_stage"] != "frames_regenerating" or not tasks:
        raise ValueError("必须先保存非空局部计划")
    return [Send("regenerate_frame", {"redo_task": task}) for task in tasks]


def make_regenerate_frame(generator: ImageGenerator):
    def regenerate_frame(state: FrameRedoInput) -> dict:
        task = state["redo_task"]
        request = ImageGenerationRequest.model_validate(task["request"])
        common = request.model_dump(mode="json")
        try:
            image = generator.generate(request)
            asset = Asset(**common, status="succeeded", image=image)
        except ImageGenerationError as exc:
            asset = Asset(**common, status="failed", error_code=exc.code)
        except (OSError, ValueError):
            asset = Asset(**common, status="failed", error_code="asset_storage_invalid")
        return {
            "assets": {request.request_id: asset.model_dump(mode="json")},
            "events": [f"局部重做完成：{task['key']} v{request.version}"],
        }

    return regenerate_frame
```
